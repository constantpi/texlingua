\section{A Simple Lemma}

\begin{lemma}\label{lem:nonnegative}
    %j 任意の $n \geq 1$ に対して、式~\eqref{eq:square} が成り立つ。
    For every $n \geq 1$, inequality~\eqref{eq:square} holds.
    \begin{equation}\label{eq:square}
        n^2 \geq 0.
    \end{equation}
\end{lemma}
\begin{proof}
    %j 実数の平方は非負である。
    The square of a real number is nonnegative.
\end{proof}
